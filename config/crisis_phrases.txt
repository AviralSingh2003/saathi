# Layer 1 of emergency detection: checked instantly, BEFORE any model sees the person's words,
# and the only layer offline. In live mode a Haiku check (host/emergency.py) catches what this
# list misses: indirect wording, context, other languages.
# Format: reason_code: regular expression (case-insensitive, matched anywhere).
# reason_code is one of: medical_emergency, self_harm, asked_for_help
# Order matters: the first matching line wins.

self_harm: \b(kill|hurt|harm)\s+myself\b
self_harm: \bend\s+(it\s+all|my\s+life)\b
self_harm: \bwant\s+to\s+die\b
self_harm: \bdon'?t\s+want\s+to\s+(live|be\s+here)\b
self_harm: \bno\s+(point|reason)\s+(in\s+)?living\b
self_harm: \bbetter\s+off\s+without\s+me\b
self_harm: \bsuicid
self_harm: \bmarna\s+chahti?\b

medical_emergency: \bseene\s+(mein|me)\s+dard\b

medical_emergency: \b(face|mouth)\s+(is\s+)?droop
